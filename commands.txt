# Person A - IID runs

# Baseline (no DP)
fluke federation config/medical-data-iid.yaml config/fl-config-svm.yaml logger.log_dir=runs/medical-SVM-iid

# DP runs with epsilon sweep.
# Smaller epsilon means stronger privacy and usually worse utility.
fluke federation config/medical-data-iid.yaml config/fl-config-svm-dp.yaml hyperparameters.client.target_epsilon=0.2 hyperparameters.client.max_grad_norm=1.0 logger.log_dir=runs/medical-dp-svm-iid-eps-0p2-mgn-1p0
fluke federation config/medical-data-iid.yaml config/fl-config-svm-dp.yaml hyperparameters.client.target_epsilon=0.5 hyperparameters.client.max_grad_norm=1.0 logger.log_dir=runs/medical-dp-svm-iid-eps-0p5-mgn-1p0
fluke federation config/medical-data-iid.yaml config/fl-config-svm-dp.yaml hyperparameters.client.target_epsilon=1.0 hyperparameters.client.max_grad_norm=1.0 logger.log_dir=runs/medical-dp-svm-iid-eps-1p0-mgn-1p0
fluke federation config/medical-data-iid.yaml config/fl-config-svm-dp.yaml hyperparameters.client.target_epsilon=5.0 hyperparameters.client.max_grad_norm=1.0 logger.log_dir=runs/medical-dp-svm-iid-eps-5p0-mgn-1p0
